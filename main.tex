\section{Introduction}

Conference styles are designed for anonymous review, not for the version people read and cite. An arXiv version needs a first page that says who wrote the paper, where its resources live, and what it finds, while keeping the reviewed body text unchanged. This template does that on top of NeurIPS-like geometry~\citep{NEURIPS2020_1457c0d6} and keeps the commands used in our recent steering papers~\citep{turner2023steering, rimsky-etal-2024-steering, zou2023transparency}.

\researchquestions{\textbf{Q1.}\enspace Can the arXiv version share the conference body text unchanged?\par\smallskip
\textbf{Q2.}\enspace Can common paper elements be styled once, in the template, instead of per paper?}

\paragraph{Running example.} In activation steering, a hidden state $\vh_t$ at generation step $t$ is shifted along a unit direction $\vr$,
\begin{equation}\label{eq:steer}
\vh_t \leftarrow \vh_t + \lambda_t \vr, \qquad \lambda_t = \lambda\,\mathbb{I}[t \le m],
\end{equation}
where full steering uses $m = T$ and prefix steering uses a short prefix $m \ll T$. \Cref{fig:schedules} plots these schedules and \Cref{alg:prefix} states the procedure.

\section{Design}

\paragraph{First page.} \texttt{\textbackslash maketitle} followed by the \texttt{abstract} environment produces the title, authors, resource rows, and abstract box; a figure with placement \texttt{[H]} right after the abstract becomes the teaser (\Cref{fig:teaser}). All resource rows (project page, code, data, models, poster, slides, video, contact) are optional: keep only the ones you have, even just one, and the rest disappear. \texttt{\textbackslash arxivaccent\{R, G, B\}} recolors the page. Review comments are on by default: \zz{a comment.} \xz{a comment.} \revise{Revised,} \edit{edited.}

\paragraph{Outline.} \texttt{\textbackslash arxivoutline} after the teaser puts the contents on its own page, and the main text starts on the next page. Its optional argument sets how much of the appendix is listed: \texttt{\textbackslash arxivoutline} lists appendix subsections, \texttt{\textbackslash arxivoutline[section]} lists only appendix sections, and \texttt{\textbackslash arxivoutline[none]} omits them. Remove the command to skip the outline page.

\paragraph{Typography.} Body text is Times at 10\,pt, headings and caption labels are Helvetica, and references are set one size smaller. Caption labels are bold sans serif and subfigure captions are one size smaller (\Cref{fig:sub}). \Cref{tab:commands} lists every command the template adds.

\begin{algorithm}[H]
\caption{Prefix steering}
\label{alg:prefix}
\begin{algorithmic}[1]
\Require prompt $x$, direction $\vr$, strength $\lambda$, $m$
\For{$t = 1, \dots, T$}
  \State $\vh_t \gets \textsc{Forward}(x, y_{<t})$
  \If{$t \le m$}
    \State $\vh_t \gets \vh_t + \lambda \vr$
  \EndIf
  \State $y_t \sim p(\cdot \mid \vh_t)$
\EndFor
\State \Return $y_{1:T}$
\end{algorithmic}
\end{algorithm}

\section{Writing Conventions}

\paragraph{Results.} State the claim, then the evidence. Use the \texttt{theorem}, \texttt{lemma}, \texttt{prop}, \texttt{cor}, \texttt{definition}, and \texttt{remark} environments, with proofs in the appendix (\appref{app:proofs}); code and prompts are in \appref{app:implementation}.

\begin{prop}\label{prop:budget}
Under \cref{eq:steer}, the total intervention $\sum_{t=1}^{T} \lambda_t \|\vr\|_2$ equals $\lambda m$, so prefix steering uses a fraction $m/T$ of the full-steering budget.
\end{prop}

\paragraph{References.} Cite with \texttt{\textbackslash citep} for parenthetical citations~\citep{chowdhery2023palm, achiam2023gpt} and \texttt{\textbackslash citet} when the authors are the subject: \citet{scalena2024multiproperty} compose several steering directions. Protect acronyms in titles with braces, such as \texttt{\{MMLU-Pro\}}~\citep{wang2024mmlu}; datasets~\citep{politeguard} and OpenReview papers~\citep{zhu2026from} use \texttt{@misc} with a \texttt{url}.

\takeaway{Write the conference version first. The arXiv version only changes the preamble: swap the style file for \texttt{arxiv.sty}, add the title-block metadata, and start the appendix with \texttt{\textbackslash arxivappendix}.}

\section{Proofs}\label{app:proofs}

\begin{proof}[Proof of \Cref{prop:budget}]
Since $\|\vr\|_2 = 1$ and $\lambda_t = \lambda$ for $t \le m$ and $0$ otherwise, $\sum_{t=1}^{T} \lambda_t \|\vr\|_2 = \sum_{t=1}^{m} \lambda = \lambda m$. Full steering has $m = T$, so the ratio is $m/T$.
\end{proof}

\begin{remark}
The budget counts only direct interventions; it ignores how a shift at step $t$ propagates to later steps through attention.
\end{remark}

\subsection{Code}

\begin{lstlisting}[language=Python]
def prefix_hook(module, inputs, output, step, r, lam, m):
    """Add lam * r to the last hidden state for the first m generated tokens."""
    if step < m:  # prefix only
        output[:, -1, :] += lam * r
    return output
\end{lstlisting}
